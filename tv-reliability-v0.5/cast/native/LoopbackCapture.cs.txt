// Windows speaker-output WASAPI loopback, no microphone capture.
// Build using Windows .NET Framework CSC 4.x (see BUILD-AUDIO.md).
// --probe emits format JSON; --capture streams that format as raw interleaved PCM to stdout.
// The service pins this executable and validates its format before streaming.
using System;
using System.IO;
using System.Runtime.InteropServices;
using System.Threading;
using System.Diagnostics;
using System.Text;
using System.Security.Cryptography;

[ComImport, Guid("A95664D2-9614-4F35-A746-DE8DB63617E6"), InterfaceType(ComInterfaceType.InterfaceIsIUnknown)]
interface IMMDeviceEnumerator {
 [PreserveSig]int EnumAudioEndpoints(int flow,int mask,out IntPtr devices);
 [PreserveSig]int GetDefaultAudioEndpoint(int flow,int role,out IMMDevice device);
 [PreserveSig]int GetDevice([MarshalAs(UnmanagedType.LPWStr)]string id,out IMMDevice device);
 [PreserveSig]int RegisterEndpointNotificationCallback(IntPtr callback);
 [PreserveSig]int UnregisterEndpointNotificationCallback(IntPtr callback);
}
[ComImport, Guid("D666063F-1587-4E43-81F1-B948E807363F"), InterfaceType(ComInterfaceType.InterfaceIsIUnknown)]
interface IMMDevice {
 [PreserveSig]int Activate(ref Guid iid,int context,IntPtr parameters,out IntPtr ppInterface);
 [PreserveSig]int OpenPropertyStore(int mode,out IntPtr props);
 [PreserveSig]int GetId(out IntPtr id);
 [PreserveSig]int GetState(out int state);
}
[ComImport, Guid("1CB9AD4C-DBFA-4C32-B178-C2F568A703B2"), InterfaceType(ComInterfaceType.InterfaceIsIUnknown)]
interface IAudioClient {
 [PreserveSig]int Initialize(int shareMode,uint flags,long bufferDuration,long periodicity,IntPtr format,ref Guid session);
 [PreserveSig]int GetBufferSize(out uint size);
 [PreserveSig]int GetStreamLatency(out long latency);
 [PreserveSig]int GetCurrentPadding(out uint padding);
 [PreserveSig]int IsFormatSupported(int mode,IntPtr format,out IntPtr closest);
 [PreserveSig]int GetMixFormat(out IntPtr format);
 [PreserveSig]int GetDevicePeriod(out long defaultPeriod,out long minimumPeriod);
 [PreserveSig]int Start();
 [PreserveSig]int Stop();
 [PreserveSig]int Reset();
 [PreserveSig]int SetEventHandle(IntPtr eventHandle);
 [PreserveSig]int GetService(ref Guid iid,out IntPtr ppInterface);
}
[ComImport, Guid("C8ADBD64-E71E-48A0-A4DE-185C395CD317"), InterfaceType(ComInterfaceType.InterfaceIsIUnknown)]
interface IAudioCaptureClient {
 [PreserveSig]int GetBuffer(out IntPtr data,out uint frames,out uint flags,out ulong devicePosition,out ulong qpcPosition);
 [PreserveSig]int ReleaseBuffer(uint frames);
 [PreserveSig]int GetNextPacketSize(out uint frames);
}
static class LoopbackCapture {
 const uint LOOPBACK=0x00020000, SILENT=2;
 const int CLSCTX_ALL=23;
 static void Check(int result) { if(result<0)Marshal.ThrowExceptionForHR(result); }
 static int U16(IntPtr ptr,int offset) { return (ushort)Marshal.ReadInt16(ptr,offset); }
 static string EndpointHash(IMMDevice device){
  IntPtr id=IntPtr.Zero;
  try{
   Check(device.GetId(out id));
   string identity=Marshal.PtrToStringUni(id);
   using(var sha=SHA256.Create()){
    var output=new StringBuilder();
    foreach(byte b in sha.ComputeHash(Encoding.Unicode.GetBytes(identity)))output.Append(b.ToString("x2"));
    return output.ToString();
   }
  }finally{if(id!=IntPtr.Zero)Marshal.FreeCoTaskMem(id);}
 }
 static string SampleFormat(IntPtr format) {
  int tag=U16(format,0), bits=U16(format,14);
  if(tag==65534) {
   if(U16(format,16)<22) return "unsupported";
   tag=Marshal.ReadInt32(format,24);
  }
  if(tag==3&&bits==32)return "f32le";
  if(tag==1&&bits==16)return "s16le";
  if(tag==1&&bits==24)return "s24le";
  if(tag==1&&bits==32)return "s32le";
  return "unsupported";
 }
 public static int Main(string[] args) {
  bool pinned=args.Length==3&&args[0]=="--capture"&&args[1]=="--device-hash"&&
   System.Text.RegularExpressions.Regex.IsMatch(args[2],@"^[0-9a-f]{64}$");
  if(!pinned&&(args.Length!=1||(args[0]!="--probe"&&args[0]!="--capture"))) {
   Console.Error.WriteLine("Usage: LoopbackCapture.exe --probe | --capture");
   return 2;
  }
  IMMDeviceEnumerator enumerator=null;
  IMMDevice device=null;
  IAudioClient client=null;
  IAudioCaptureClient capture=null;
  IntPtr format=IntPtr.Zero;
  bool started=false;
  try {
   // eRender (0) / eMultimedia (1): only speaker-output, never eCapture/microphone.
   Type type=Type.GetTypeFromCLSID(new Guid("BCDE0395-E52F-467C-8E3D-C4579291692E"));
   enumerator=(IMMDeviceEnumerator)Activator.CreateInstance(type);
   Check(enumerator.GetDefaultAudioEndpoint(0,1,out device));
   string playbackHash=EndpointHash(device);
   if(pinned&&playbackHash!=args[2])throw new InvalidOperationException("Windows playback endpoint changed after HLS audio probe");
   Guid cid=typeof(IAudioClient).GUID;
   IntPtr p;Check(device.Activate(ref cid,CLSCTX_ALL,IntPtr.Zero,out p));
   client=(IAudioClient)Marshal.GetObjectForIUnknown(p);Marshal.Release(p);
   Check(client.GetMixFormat(out format));
   int rate=Marshal.ReadInt32(format,4),channels=U16(format,2);
   int align=U16(format,12),bits=U16(format,14);
   string sample=SampleFormat(format);
   bool valid=sample!="unsupported"&&channels>=1&&channels<=8&&
      rate>=8000&&rate<=192000&&bits%8==0&&align==channels*(bits/8);
   if(args[0]=="--probe") {
    Console.WriteLine("{\"ok\":"+ (valid?"true":"false")+
      ",\"sampleFormat\":\""+sample+"\",\"sampleRate\":"+rate+
      ",\"channels\":"+channels+",\"bitsPerSample\":"+bits+",\"blockAlign\":"+align+
      ",\"deviceIdHash\":\""+playbackHash+"\""+
      ",\"capture\":\"default-speaker-output-loopback\"}");
    return valid?0:4;
   }
   if(!valid)throw new InvalidOperationException("Unexpected Windows speaker mix format");
   Guid session=Guid.Empty;
   Check(client.Initialize(0,LOOPBACK,1000000,0,format,ref session));
   Guid aid=typeof(IAudioCaptureClient).GUID;
   IntPtr q;Check(client.GetService(ref aid,out q));
   capture=(IAudioCaptureClient)Marshal.GetObjectForIUnknown(q);Marshal.Release(q);
   Check(client.Start());started=true;
   Console.Error.WriteLine("LOOPBACK_READY "+sample+" "+rate+" "+channels);
   // Important: no device data ever written to disk; stdout only.
   using(Stream output=Console.OpenStandardOutput()) {
    Stopwatch clock=Stopwatch.StartNew();
    long framesWritten=0;
    int silenceFrames=Math.Max(1,rate/100);
    byte[] zero=new byte[silenceFrames*align];
    while(true){
     uint pending;Check(capture.GetNextPacketSize(out pending));
     while(pending>0){
      IntPtr address;uint frames,flags;ulong devicePosition,qpcPosition;
      Check(capture.GetBuffer(out address,out frames,out flags,out devicePosition,out qpcPosition));
      try{
       int n=checked((int)frames*align);
       byte[] data=new byte[n];
       if(n>0&&(flags&SILENT)==0)Marshal.Copy(address,data,0,n);
       output.Write(data,0,n);
       framesWritten+=frames;
      } finally {Check(capture.ReleaseBuffer(frames));}
      Check(capture.GetNextPacketSize(out pending));
     }
     // Some WASAPI endpoints suspend packets when nothing plays. Feed bounded silence
     // to preserve HLS audio timestamps, not a second desktop/microphone recording.
     long expected=(long)(clock.Elapsed.TotalSeconds*rate);
     long deficit=expected-framesWritten;
     if(deficit>=rate/15){
      int fill=(int)Math.Min(deficit-rate/40,silenceFrames);
      if(fill>0){output.Write(zero,0,fill*align);framesWritten+=fill;}
     }
     Thread.Sleep(8);
    }
   }
  }catch(IOException){return 0;} // owner session closed PCM pipe
   catch(Exception e){
    Console.Error.WriteLine("LOOPBACK_FAILED "+e.GetType().Name+": "+e.Message);
    return 5;
   }finally{
    if(started)try{client.Stop();}catch{}
    if(format!=IntPtr.Zero)Marshal.FreeCoTaskMem(format);
    if(capture!=null)Marshal.ReleaseComObject(capture);
    if(client!=null)Marshal.ReleaseComObject(client);
    if(device!=null)Marshal.ReleaseComObject(device);
    if(enumerator!=null)Marshal.ReleaseComObject(enumerator);
   }
 }
}
